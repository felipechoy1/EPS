\section{Introducción}
%------------------------------------------------

\begin{frame}{Introducción}
\vspace{-0.4em}
\begin{itemize}
    \item En la computación cuántica, modelar cómo evoluciona un sistema abierto es fundamental.
    
    \medskip
    \item Esto se describe mediante \textbf{canales cuánticos}: operadores que representan la interacción del sistema con su entorno.

    \medskip
    \item Sin embargo, los canales cuánticos ideales son difíciles de implementar en dispositivos reales debido al \textbf{ruido cuántico}.

    \medskip
    \item ¿Cómo podemos implementar y validar estos canales en una computadora cuántica actual?
\end{itemize}
\end{frame}

\section{Conceptos Fundamentales}
%------------------------------------------------


\begin{frame}{Conceptos Fundamentales}
\framesubtitle{Matriz Densidad}

\begin{columns}
    \column{0.60\textwidth}
    \begin{itemize}
        \item Un estado cuántico puro se representa por un vector $\ket{\psi}$ en un espacio de Hilbert.

        \medskip
        \item Si el sistema no está en un solo estado, usamos una combinación estadística: un estado mixto.

        \medskip
        \item Ambos casos se describen con la \textbf{matriz densidad} $\rho$.

        \medskip
        \item Cuando el sistema está compuesto por dos partes ($A$ y $B$), podemos usar la \textbf{traza parcial} para estudiar solo una de ellas.

    \end{itemize}

    \column{0.42\textwidth}
    \bigskip
    $$
    \ket{\psi} = \sum_i c_i \ket{i}
    $$
    \bigskip
    $$
    \rho =\begin{cases}
        \ket{\psi}\bra{\psi} & \text{(puro)} \\
        \sum_i p_i \ket{\psi_i} \bra{\psi_i} & \text{(mixto)}
    \end{cases}
    $$
    \medskip
    $$
    \rho_{AB} = \rho_A \otimes \rho_B
    $$
    $$
    \rho_A = \text{Tr}_B(\rho_{AB})
    $$
    $$
    \rho_A = \ket{i_A} \bra{j_A} \text{Tr}(\ket{i_B}\bra{j_B})
    $$
\end{columns}
\end{frame}

\begin{frame}{Conceptos Fundamentales}
\framesubtitle{Canales Cuánticos}

\begin{columns}
    \column{0.60\textwidth}
    \begin{itemize}
        \item Un \textbf{canal cuántico} $\mathcal{E}$ describe cómo evoluciona un estado cuántico, incluyendo interacciones con el entorno (ruido).

        \medskip
        \item Estos canales deben ser:
        \begin{itemize}
            \item \textbf{Completamente positivos (CP)}
            \item \textbf{Preservadores de la traza (TP)}
        \end{itemize}

        \bigskip
        \item Se pueden representar mediante una \textbf{matriz de Choi}, que es única para cada canal.
    \end{itemize}

    \column{0.42\textwidth}
    \vspace{2 em}
    $$
    \mathcal{E}(\rho) \rightarrow \rho'
    $$
    % \smallskip
    $$
    \Lambda_{\mathcal{E}} = \sum_{i, j=1}^{d} \ketbra{i}{j} \otimes \mathcal{E}(\ketbra{i}{j})
    $$
    $$
    \mathcal{E} (\rho) = \text{Tr}_1 [ \Lambda (\rho^{T} \otimes I)]
    $$
    \begin{itemize}
        \item $\Lambda_{\mathcal{E}}$ es positiva semidefinida para un mapeo CP. 
        \item $\mathrm{Tr}_2(\Lambda) = I$ ssi es TP.
    \end{itemize}
\end{columns}
\end{frame}

%======================================
\begin{frame}{Conceptos Fundamentales}
\framesubtitle{Circuitos Cuánticos}

\begin{columns}
\column{0.50\textwidth}
\begin{itemize}
    \item Un circuito cuántico es una secuencia de operaciones sobre uno o más \textbf{qubits}, representados por líneas horizontales.

    \medskip
    \item Las \textbf{compuertas cuánticas} son operadores que modifican el estado del qubit, se representan por 'cajas'.

    \medskip
    \item Las compuertas que involucran múltiples qubits, son \textbf{compuertas de control}.

    \medskip
    \item Una \textbf{medición} colapsa el estado cuántico en un resultado clásico.
\end{itemize}

\column{0.42\textwidth}
\begin{figure}[h]
    \centering
    \begin{quantikz}
    \lstick{$\ket{0}$} & \gate{X} & \ctrl{1} & \qw  & \meter{} \\
    \lstick{$\ket{0}$} & \qw & \gate{R_y} & \qw  & \qw
    \end{quantikz}
\end{figure}

\smallskip
\begin{itemize}
    \item[] \begin{quantikz}
       \lstick{$\ket{\psi}$} & \qw
    \end{quantikz} = qubit
    \item[] \begin{quantikz}
        \gate{X}
    \end{quantikz} = compuerta
    \item[] $\bullet$ = control
    \item[] \begin{quantikz}
        \meter{}
    \end{quantikz} = medición
\end{itemize}

\end{columns}
\end{frame}
%=========================================



\begin{frame}{Conceptos Fundamentales}
\framesubtitle{Tomografía y Fidelidad de Proceso}

\begin{columns}
    \column{0.65\textwidth}
    \begin{itemize}
        \item La \textbf{tomografía de proceso} permite reconstruir un canal cuántico $\mathcal{E}$ a partir de mediciones sobre diferentes estados de entrada.

        \medskip
        \item Las salidas $\mathcal{E}(\ketbra{\psi_j}{\psi_j})$ permiten reconstruir $\Lambda_{\mathcal{E}}$.

        \medskip
        \item La \textbf{fidelidad de proceso} compara la matriz de Choi reconstruida con la ideal:
    \end{itemize}

    \begin{equation*}
    \begin{aligned}
    F_{\text{proc}} &= \frac{1}{d^2}F(\Lambda_{\mathcal{E}}, \Lambda_{\mathcal{F}}), \\
    \text{donde } F(\Lambda_{\mathcal{E}}, \Lambda_{\mathcal{F}}) &= \text{Tr} \sqrt{\Lambda_{\mathcal{E}}^{1/2} \Lambda_{\mathcal{F}} \Lambda_{\mathcal{E}}^{1/2} } 
    \end{aligned}
    \end{equation*}

    \column{0.30\textwidth}
    
    \vspace{0.5em}
    \begin{quantikz}
    \lstick{$\ket{0}$}  & \gate{\mathcal{E}} & \qw & \rstick{$\rho_1$} \\
    \lstick{$\ket{1}$}  & \gate{\mathcal{E}} & \qw & \rstick{$\rho_2$} \\
    \lstick{$\ket{+}$}  & \gate{\mathcal{E}} & \qw & \rstick{$\rho_3$} \\
    \lstick{$\ket{+i}$} & \gate{\mathcal{E}} & \qw & \rstick{$\rho_4$}
    \end{quantikz}

    \vspace{1em}
    \small
    $\rho_j = \mathcal{E}(\ketbra{\psi_j}{\psi_j})$
\end{columns}
\end{frame}

\section{Algoritmos Variacionales Cuánticos}
%------------------------------------------------

\begin{frame}{Algoritmos Variacionales Cuánticos}
\begin{itemize}
    \item Combinan computación cuántica y optimización clásica para resolver tareas específicas.
    
    \item Se basan en un circuito cuántico parametrizado $U(\bm{\theta})$ aplicado sobre un estado inicial $\ket{\psi_0}$:
    \begin{equation*}
    \ket{\psi(\bm{\theta})} = U(\bm{\theta}) \ket{\psi_0},   
    \end{equation*}    
    donde $\bm{\theta}$ es optimizado para minimizar una función de costo $C(\bm{\theta})$.
    
    \item Buscan:
    \begin{equation*}
        \bm{\theta}^* = \underset{\bm{\theta}}{\text{arg min}}\, C(\bm{\theta}).        
    \end{equation*}
\end{itemize}
\end{frame}

%-------------------------------------------------
\begin{frame}{Algoritmos Variacionales Cuánticos}
\framesubtitle{Hardware Efficient Ansatz (HEA)}
\begin{columns}[c]
\column{.60\textwidth} 
    \begin{itemize}
        \item El \textit{ansatz} define la estructura de $U(\bm{\theta})$ , así como el tipo de compuertas cuánticas.
       
        \item El HEA usa compuertas nativas del dispositivo cuántico, además son agnósticos al problema a resolver.
       
        \item Escogimos el HEA:
        \begin{equation*}
        % \label{ec:ansatz}
             U(\bm{\theta}) =\prod_{i=1}^{M} \prod_{j=1}^N U_{ent} (\theta_i)R_z(\theta_{ij})R_x(\theta_{ij2})R_z(\theta_{ij3}).
         \end{equation*}
    \end{itemize}

\section{Resultados para el Canal Despolarizante de un qubit}

\begin{frame}{Resultados}
\framesubtitle{Canal Despolarizante}
    \begin{columns}[c] 

        \column{.60\textwidth} % Left column and width
    \begin{itemize}
        \item Modelo de ruido cuántico con probabilidad $p$ de convertir $\rho$ en $\frac{{I}}{2}$ mediante 
        \begin{equation*}
            \mathcal{E(\rho)}= \text{Tr}(\rho) p \frac{I}{2} + (1-p) \rho.
            \end{equation*}
  \item {} 
  \begin{equation*}
  % \label{eq:choi_depolarizing}
    \Lambda_{\mathcal E}=\Lambda_{\text{obj}}=
    \begin{pmatrix}
      1-\frac{p}{2} & 0 & 0 & 1-p \\
      0 & \frac{p}{2} & 0 & 0 \\
      0 & 0 & \frac{p}{2} & 0 \\
      1-p & 0 & 0 & 1-\frac{p}{2}
    \end{pmatrix}.
  \end{equation*}
    \end{itemize}

\section{Conclusiones y trabajo a futuro}
%------------------------------------------------



\begin{frame}{Conclusiones}
\begin{itemize}
    \item Implementamos el canal despolarizante de un qubit usando algoritmos variacionales cuánticos (VQA), mediante un \textit{hardware-efficient ansatz} optimizado por fidelidad de proceso.
    
    \item Las ejecuciones en simulador cuántico mostraron fidelidades superiores a 0.900 con baja variabilidad, confirmando la efectividad del enfoque en entornos ideales.
    
    \item En la computadora cuántica (\texttt{ibm\_kyoto}), observamos una tendencia creciente en la fidelidad conforme aumentó la probabilidad de despolarización $p$, con una regresión lineal de pendiente $m = 0.2707$ y $R^2 = 0.7055$.
    
    \item La reducción de variabilidad con $p$ sugiere mayor tolerancia al ruido en estados más mixtos; sin embargo, las matrices de Choi indicaron que el hardware preserva coherencias no triviales en bajas despolarizaciones.
    
    % \item El estudio demuestra que los VQA son una herramienta prometedora para simular canales cuánticos en hardware cuántico, aunque su desempeño está condicionado por el ruido y la calidad del dispositivo.
\end{itemize}
\end{frame}
